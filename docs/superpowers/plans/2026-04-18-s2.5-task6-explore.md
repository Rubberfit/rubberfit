Loop structure (Algorithm 12, BPP single-bin adaptation):

```
// Pool is sorted by loss ascending; best (smallest loss) at index 0.
let mut pool: Vec<(LayoutSnapshot, f32)> = Vec::new();

// Explicit counter, independent of pool.len(). Under early-exit they
// happen to coincide today, but naming parity with sparrow's
// config.max_conseq_failed_attempts demands the real invariant
// (Gemini sign-off R1).
let mut conseq_failed: usize = 0;

while !term.kill() {
    match sep.separate(term, listener) {
        Feasible(snap) => {
            // Early exit — per-bin density is fixed by item set; no value
            // in further exploration. Matches BPP reality + Codex sign-off.
            listener.report_feasible(&snap);
            return ExplorationOutcome::Feasible(snap);
        }
        Infeasible { snapshot, ct_snapshot: _ } => {
            let loss = sep.ct.get_total_loss();
            debug_assert!(loss.is_finite(), "infeasible pool must not receive NaN/inf loss");
            listener.report_infeasible(&snapshot);
            insert_sorted_by_loss(&mut pool, (snapshot, loss));
            conseq_failed += 1;
            if conseq_failed >= config.max_conseq_failed_attempts.unwrap_or(usize::MAX) {
                break;
            }
            let selected = sample_pool_biased(&pool, &mut sep.rng,
                                              config.solution_pool_distribution_stddev);
            // Fresh tracker (upstream L79 uses None — discards weights on
            // disrupt; preserves the paper's GLS escape behaviour).
            sep.rollback(&selected.0, None);
            disrupt_solution(sep, config);
        }
    }
}

// Fell out of loop (terminator or attempts cap). Return best-infeasible
// (pool head if any, else current layout).
match pool.into_iter().next() {
    Some((snap, loss)) => ExplorationOutcome::Infeasible { snapshot: snap, total_loss: loss },
    None => ExplorationOutcome::Infeasible {
        snapshot:   sep.layout.save(),
        total_loss: sep.ct.get_total_loss(),
    },
}
```

### 3.3 `disrupt_solution`

Port of sparrow L89-216. Key adaptation per Codex #3: **large-item cutoff computed from `sep.layout.placed_items` (BPP: this bin only), not `sep.instance.items` (BPP: full demand across bins)**.

```rust
fn disrupt_solution(sep: &mut Separator, config: &ExplorationConfig) {
    if sep.layout.placed_items.len() < 2 {
        warn!("[DSRP] cannot disrupt solution with less than 2 items");
        return;
    }

    // BPP deviation: compute large-item cutoff from items CURRENTLY PLACED
    // in this layout, not from instance demand totals. For SPP the two are
    // identical (single strip holds everything); for BPP the instance
    // holds items destined for other bins too.
    let total_ch_area: f32 = sep
        .layout
        .placed_items
        .values()
        .map(|pi| sep.instance.item(pi.item_id).shape_cd.surrogate().convex_hull_area)
        .sum();
    let cutoff_threshold_area = total_ch_area * config.large_item_ch_area_cutoff_percentile;

    // Sort placed items by ch area desc. Walk until cumulative > threshold;
    // the item that tipped the sum defines the cutoff.
    let mut by_area: Vec<(PItemKey, f32)> = sep
        .layout
        .placed_items
        .iter()
        .map(|(pk, pi)| (pk, sep.instance.item(pi.item_id).shape_cd.surrogate().convex_hull_area))
        .collect();
    by_area.sort_by_key(|(_, a)| Reverse(OrderedFloat(*a)));

    let mut cumulative = 0.0;
    let mut ch_area_cutoff = 0.0;
    for &(_, a) in &by_area {
        cumulative += a;
        if cumulative > cutoff_threshold_area {
            ch_area_cutoff = a;
            break;
        }
    }
    // Safety rail (Gemini sign-off nice-to-have): if the accumulation loop
    // never tips the threshold, `ch_area_cutoff` stays 0.0 and the
    // subsequent "large item" filter catches everything. Not silently
    // wrong, but worth guarding in debug builds.
    debug_assert!(
        ch_area_cutoff > 0.0,
        "large-item cutoff should be positive; placed_items={}",
        sep.layout.placed_items.len()
    );

    // Pick two distinct "large" items. Second must differ from the
    // first in BOTH area AND diameter by >1% (upstream uses && at
    // explore.rs L147-152 — not OR). Fall back to any other placed
    // item if that filter finds nothing.
    let large: Vec<PItemKey> = sep
        .layout
        .placed_items
        .iter()
        .filter(|(_, pi)| pi.shape.surrogate().convex_hull_area >= ch_area_cutoff)
        .map(|(pk, _)| pk)
        .collect();
    let pk1 = *large.choose(&mut sep.rng).expect("[DSRP] no first large item");
    let pi1 = &sep.layout.placed_items[pk1];
    let shape1_area = pi1.shape.area;
    let shape1_diam = pi1.shape.diameter;

    let pk2 = large
        .iter()
        .copied()
        .filter(|&pk| pk != pk1)
        .filter(|&pk| {
            let pi = &sep.layout.placed_items[pk];
            !approx_eq!(f32, pi.shape.area, shape1_area, epsilon = shape1_area * 0.01)
                && !approx_eq!(f32, pi.shape.diameter, shape1_diam, epsilon = shape1_diam * 0.01)
        })
        .choose(&mut sep.rng)
        .or_else(|| {
            sep.layout
                .placed_items
                .keys()
                .filter(|&pk| pk != pk1)
                .choose(&mut sep.rng)
        })
        .expect("[DSRP] no second item");

    // Swap positions (with rotation-feasible adjustment via
    // convert_sample_to_closest_feasible, already ported in
    // uniform_sampler.rs).
    let dt1_old = sep.layout.placed_items[pk1].d_transf;
    let dt2_old = sep.layout.placed_items[pk2].d_transf;
    let item1 = sep.instance.item(sep.layout.placed_items[pk1].item_id);
    let item2 = sep.instance.item(sep.layout.placed_items[pk2].item_id);
    let dt1_new = convert_sample_to_closest_feasible(dt2_old, item1);
    let dt2_new = convert_sample_to_closest_feasible(dt1_old, item2);

    let pk1 = sep.move_item(pk1, dt1_new);
    let pk2 = sep.move_item(pk2, dt2_new);

    // Relocate items practically contained by each swapped item.
    relocate_contained(sep, pk1, dt1_old, dt1_new, pk2);
    relocate_contained(sep, pk2, dt2_old, dt2_new, pk1);
}

### 3.4 Restored `Separator::move_item`

/// Move a placed item to a new transformation. Returns the new
/// `PItemKey` issued by `Layout::place_item` (Layout emits a fresh key
/// on every placement). Intended for call from the explore.rs disrupt
/// path — Task 5 initially dropped this method; Task 6 re-adds it.
pub fn move_item(&mut self, pk: PItemKey, d_transf: DTransformation) -> PItemKey {
    let item_id = self.layout.placed_items[pk].item_id;
    let item = self.instance.item(item_id);
    self.layout.remove_item(pk);
    let new_pk = self.layout.place_item(item, d_transf);
    self.ct.register_item_move(&self.layout, pk, new_pk);
    new_pk
}
```
