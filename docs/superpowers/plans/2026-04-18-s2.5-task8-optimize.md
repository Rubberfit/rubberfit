### 3.2 `search::optimize` body

```rust
pub fn optimize(
    seed: SeedResult,
    config: OptimizerConfig,
) -> Result<SeedResult, PackError> {
    // 1. Build BPInstance from domain context carried in config.
    let instance = jagua_adapter::to_jagua_instance(
        std::slice::from_ref(&config.bin),
        &config.items,
        config.clearance,
    )?;

    // 2. Construct a Layout and replay warm-start placements into it.
    let container = instance.container(0).clone();
    let mut layout = Layout::new(container);
    replay_seed_into_layout(&mut layout, &instance, &seed)?;

    // 3. Build a Separator from (instance, layout, rng, SeparatorConfig::exploration()).
    let rng = search::rng::from_seed(config.seed);
    let mut sep = Separator::new(instance, layout, rng, SeparatorConfig::exploration());

    // 4. Arm a BasicTerminator. 100% explore, compress is identity.
    let term = BasicTerminator::with_timeout(Duration::from_millis(config.time_budget_ms));
    let mut listener = NoopListener;

    // 5. Run exploration -> compression (identity).
    let expl_outcome = exploration_phase(
        &mut sep, &term, &mut listener, &ExplorationConfig::paper_defaults(),
    );
    let final_snap = match expl_outcome {
        ExplorationOutcome::Feasible(snap) => snap,
        ExplorationOutcome::Infeasible { snapshot, .. } => snapshot,
    };
    let polished = compression_phase(&mut sep, final_snap, &term, &mut listener);

    // 6. Translate back to SeedResult.
    Ok(snapshot_to_seed(&polished, &sep.instance, &config.items, &config.bin.bin_id))
}
```

### 3.3 `replay_seed_into_layout` helper

Take the warm-start `SeedResult`'s `SeedPlacement` entries, map each back to `(item_id, DTransformation)`, and call `layout.place_item(item, d_transf)`. The warm-start already carries `(x, y)` + `rotation_rad` per placement (see `warm_start.rs:56-69`).

**Lookup convention (Codex sign-off fix): use an explicit `HashMap<String, usize>` keyed by `shape_id`. Do not rely on positional matching or per-placement linear scans.** Build the map once from `config.items`, assert no duplicate `shape_id`, and error on missing lookups. This is more robust than assuming seed order matches config order forever.

```rust
fn replay_seed_into_layout(
    layout: &mut Layout,
    instance: &BPInstance,
    items: &[Part],
    seed: &SeedResult,
) -> Result<(), PackError> {
    // Build shape_id -> item_id lookup once. Error on duplicate shape_ids
    // (the jagua adapter already enforces this upstream but defence-in-
    // depth is cheap).
    let mut shape_to_item_id: HashMap<&str, usize> = HashMap::with_capacity(items.len());
    for (idx, part) in items.iter().enumerate() {
        if shape_to_item_id.insert(part.shape_id.as_str(), idx).is_some() {
            return Err(PackError::EngineFailed(format!(
                "duplicate shape_id '{}' in items — refusing to replay seed",
                part.shape_id
            )));
        }
    }

    for placement in &seed.placements {
        let item_id = *shape_to_item_id.get(placement.shape_id.as_str()).ok_or_else(|| {
            PackError::EngineFailed(format!(
                "seed placement references unknown shape_id '{}'",
                placement.shape_id
            ))
        })?;
        let item = instance.item(item_id);
        let dt = DTransformation::new(
            placement.rotation_rad as f32,
            (placement.x as f32, placement.y as f32),
        );
        layout.place_item(item, dt);
    }
    Ok(())
}
```

Rotation: jagua requires radians (f32); `SeedPlacement.rotation_rad` is f64 — narrow via `x as f32` (the f64→f32 cast is safe for our rotation range).

### 3.4 `snapshot_to_seed` helper

```rust
fn snapshot_to_seed(
    snap: &LayoutSnapshot,
    instance: &BPInstance,
    items: &[Part],
    bin_id: &str,
) -> SeedResult {
    // Step 1: collect placements in deterministic order.
    let mut entries: Vec<(&PlacedItem,)> = snap
        .placed_items
        .values()
        .map(|pi| (pi,))
        .collect();
    entries.sort_by_key(|(pi,)| {
        // (item_id, translation.0, translation.1, rotation) for stable
        // numbering under fixed seed. f32 ordered via to_bits avoids
        // NaN issues (f32 has no NaN for valid placements).
        (
            pi.item_id,
            pi.d_transf.translation.0.into_inner().to_bits(),
            pi.d_transf.translation.1.into_inner().to_bits(),
            pi.d_transf.rotation.into_inner().to_bits(),
        )
    });

    // Step 2: translate each placement. Use jagua's int_to_ext_transformation
    // (io::export.rs:34) so the DTransformation is correctly composed with
    // the pre_transform jagua applied during import.
    let mut per_shape_count: HashMap<usize, u32> = HashMap::new();
    let mut placements = Vec::with_capacity(entries.len());
    for (pi,) in entries {
        let item = instance.item(pi.item_id);
        let pre_transf = &item.shape_orig.pre_transform;
        let abs_transf = int_to_ext_transformation(&pi.d_transf, pre_transf);
        // Apply abs_transf to the ORIGINAL input polygon (from config.items)
        let part = &items[pi.item_id];
        let polygon: Vec<Point2D> = part.polygon
            .iter()
            .map(|p| apply_dtransformation_to_point(abs_transf, *p))
            .collect();

        let idx = per_shape_count.entry(pi.item_id).or_insert(0);
        let instance_index = *idx;
        *idx += 1;

        let (tx, ty) = (abs_transf.translation.0.into_inner(), abs_transf.translation.1.into_inner());
        let rotation = abs_transf.rotation.into_inner();

        // bbox of the transformed polygon
        let (w, h) = bbox_wh(&polygon);

        placements.push(SeedPlacement {
            shape_id: part.shape_id.clone(),
            instance_index,
            bin_id: bin_id.to_string(),
            x: tx as f64,
            y: ty as f64,
            rotation_rad: rotation as f64,
            polygon,
            width: w,
            height: h,
        });
    }

    // Step 3: overflow. Per-shape: requested - placed.
    let total_requested: u32 = items.iter().map(|p| p.quantity).sum();
    let total_placed = placements.len() as u32;
    let overflow = total_requested.saturating_sub(total_placed);

    SeedResult { placements, overflow }
}

fn apply_dtransformation_to_point(dt: DTransformation, p: Point2D) -> Point2D {
    // Apply rotation then translation (jagua's DTransformation is decomposed
    // as: rotate about origin, then translate).
    let cos = dt.rotation.into_inner().cos();
    let sin = dt.rotation.into_inner().sin();
    let (px, py) = (p.x as f32, p.y as f32);
    let rx = px * cos - py * sin + dt.translation.0.into_inner();
    let ry = px * sin + py * cos + dt.translation.1.into_inner();
    Point2D::new(rx as f64, ry as f64)
}
```
